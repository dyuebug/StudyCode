## 🔥 核心知识点速览

```rust
// 切片语法
let s = String::from("hello world");
let hello = &s[0..5];   // "hello"
let world = &s[6..];    // "world"
let all   = &s[..];     // "hello world"

// 数组切片
let a = [1, 2, 3, 4, 5];
let slice = &a[1..3];   // [2, 3]

// 生命周期注解
fn longest<'a>(x: &'a str, y: &'a str) -> &'a str {
    if x.len() > y.len() { x } else { y }
}

// 结构体中的生命周期
struct Excerpt<'a> {
    part: &'a str,
}
```
