### 任务1：理解 RAII 原理（预计30分钟）

**什么是 RAII？**

RAII（Resource Acquisition Is Initialization）是 C++ 中管理资源的重要技术：
- 资源在对象构造时获取
- 资源在对象析构时释放
- 利用 C++ 的自动析构机制

**为什么需要 RAII？**

传统的手动内存管理问题：
```cpp
// 问题代码
void processData() {
    int* data = new int[1000];
    
    // 如果这里抛出异常，内存泄漏！
    if (someCondition) {
        return;  // 忘记 delete，内存泄漏！
    }
    
    // 处理数据...
    
    delete[] data;  // 可能永远执行不到
}
```

**RAII 解决方案**：
```cpp
// 使用 RAII
void processData() {
    std::vector<int> data(1000);  // 自动管理内存
    
    if (someCondition) {
        return;  // 没问题，vector 会自动释放内存
    }
    
    // 处理数据...
    
    // 函数结束时，vector 自动释放内存
}
```

**RAII 的优势**：
1. 自动资源管理，不会忘记释放
2. 异常安全，即使抛出异常也能正确释放
3. 代码更简洁，不需要手动 delete

### 任务2：学习 unique_ptr（预计40分钟）

**什么是 unique_ptr？**

`unique_ptr` 是独占所有权的智能指针：
- 同一时刻只能有一个 unique_ptr 指向对象
- 不能复制，只能移动
- 对象销毁时自动释放内存

**基本用法**：

```cpp
#include <memory>
#include <iostream>
using namespace std;

class Student {
private:
    string name;
public:
    Student(const string& n) : name(n) {
        cout << "Student " << name << " 创建" << endl;
    }
    
    ~Student() {
        cout << "Student " << name << " 销毁" << endl;
    }
    
    void display() const {
        cout << "姓名：" << name << endl;
    }
};

int main() {
    // 创建 unique_ptr
    unique_ptr<Student> ptr1(new Student("张三"));
    
    // 更好的方式：使用 make_unique（C++14）
    auto ptr2 = make_unique<Student>("李四");
    
    // 使用智能指针
    ptr1->display();
    ptr2->display();
    
    // 不能复制
    // unique_ptr<Student> ptr3 = ptr1;  // 错误！
    
    // 可以移动
    unique_ptr<Student> ptr3 = move(ptr1);
    // 现在 ptr1 为空，ptr3 拥有对象
    
    if (ptr1 == nullptr) {
        cout << "ptr1 已经为空" << endl;
    }
    
    // 函数结束时，所有对象自动销毁
    return 0;
}
```

**unique_ptr 的常用操作**：

```cpp
// 1. 创建
auto ptr = make_unique<Student>("张三");

// 2. 访问对象
ptr->display();        // 通过 -> 访问成员
(*ptr).display();      // 通过 * 解引用

// 3. 获取原始指针
Student* raw = ptr.get();

// 4. 释放所有权
Student* raw2 = ptr.release();  // ptr 变为空，需要手动 delete raw2
delete raw2;

// 5. 重置指针
ptr.reset();                    // 释放当前对象，ptr 变为空
ptr.reset(new Student("王五")); // 释放当前对象，指向新对象

// 6. 检查是否为空
if (ptr) {
    cout << "ptr 不为空" << endl;
}
```

**unique_ptr 数组**：

```cpp
// 管理数组
auto arr = make_unique<int[]>(10);
arr[0] = 100;
arr[1] = 200;

// 自动释放数组内存
```

#include <iostream>
#include <vector>

class Resource {

unique_ptr<Resource> createResource(const string& name, int id) {
    return make_unique<Resource>(name, id);

// 函数接受 unique_ptr（转移所有权）

}

    // 1. 创建和使用
    {
        cout << "\n【示例1】创建和使用" << endl;
        auto res1 = make_unique<Resource>("资源1", 1);

    // 2. 移动语义
    {
        cout << "\n【示例2】移动语义" << endl;
        auto res2 = make_unique<Resource>("资源2", 2);
        auto res3 = move(res2);  // 转移所有权
        
        if (res2 == nullptr) {
            cout << "res2 已经为空" << endl;
        }
        
        res3->use();
    }
    
    // 3. 函数返回
    {
        cout << "\n【示例3】函数返回" << endl;
        auto res4 = createResource("资源4", 4);
        res4->use();
    }
    
    // 4. 函数参数（转移所有权）
    {
        cout << "\n【示例4】函数参数" << endl;
        auto res5 = make_unique<Resource>("资源5", 5);
        processResource(move(res5));
        
        if (res5 == nullptr) {
            cout << "res5 已经为空（所有权已转移）" << endl;
        }
    }
    
    // 5. 容器中使用
    {
        cout << "\n【示例5】容器中使用" << endl;
        vector<unique_ptr<Resource>> resources;
        
        resources.push_back(make_unique<Resource>("资源6", 6));
        resources.push_back(make_unique<Resource>("资源7", 7));
        resources.push_back(make_unique<Resource>("资源8", 8));
        
        for (const auto& res : resources) {
            res->use();
        }
    }  // 容器销毁时，所有资源自动释放
    
    cout << "\n程序结束" << endl;
    return 0;
}

### 任务3：学习 shared_ptr（预计40分钟）

**什么是 shared_ptr？**

`shared_ptr` 是共享所有权的智能指针：
- 多个 shared_ptr 可以指向同一个对象
- 使用引用计数管理对象生命周期
- 最后一个 shared_ptr 销毁时，对象才被释放

**基本用法**：

```cpp
#include <memory>
#include <iostream>
using namespace std;

class Data {
private:
    int value;
public:
    Data(int v) : value(v) {
        cout << "Data(" << value << ") 创建" << endl;
    }
    
    ~Data() {
        cout << "Data(" << value << ") 销毁" << endl;
    }
    
    int getValue() const { return value; }
};

int main() {
    // 创建 shared_ptr
    shared_ptr<Data> ptr1 = make_shared<Data>(100);
    cout << "引用计数：" << ptr1.use_count() << endl;  // 1
    
    // 复制 shared_ptr（引用计数增加）
    shared_ptr<Data> ptr2 = ptr1;
    cout << "引用计数：" << ptr1.use_count() << endl;  // 2
    
    {
        shared_ptr<Data> ptr3 = ptr1;
        cout << "引用计数：" << ptr1.use_count() << endl;  // 3
    }  // ptr3 销毁，引用计数减1
    
    cout << "引用计数：" << ptr1.use_count() << endl;  // 2
    
    // 最后一个 shared_ptr 销毁时，对象才被释放
    return 0;
}
```

**shared_ptr 的常用操作**：

```cpp
// 1. 创建
auto ptr = make_shared<Data>(100);

// 2. 复制（引用计数增加）
auto ptr2 = ptr;

// 3. 获取引用计数
cout << ptr.use_count() << endl;

// 4. 检查是否唯一
if (ptr.unique()) {
    cout << "只有一个引用" << endl;
}

// 5. 重置
ptr.reset();  // 引用计数减1，如果为0则释放对象

// 6. 获取原始指针
Data* raw = ptr.get();
```

#include <memory>

#include <vector>

// 共享资源的示例

private:
    shared_ptr<Node> node;

    SharedResource(shared_ptr<Node> n) : node(n) {
        cout << "SharedResource 创建，引用计数：" << node.use_count() << endl;
    }
    
    void display() const {
        cout << "数据：" << node->getData() 
             << "，引用计数：" << node.use_count() << endl;
    }
};

int main() {
    cout << "=== shared_ptr 基础示例 ===" << endl;
    
    // 1. 基本使用
    {
        cout << "\n【示例1】基本使用" << endl;
        auto ptr1 = make_shared<Node>(100);
        cout << "引用计数：" << ptr1.use_count() << endl;
        
        auto ptr2 = ptr1;
        cout << "引用计数：" << ptr1.use_count() << endl;
        
        auto ptr3 = ptr1;
        cout << "引用计数：" << ptr1.use_count() << endl;
    }  // 所有 shared_ptr 销毁，对象才被释放
    
    // 2. 共享资源
    {
        cout << "\n【示例2】共享资源" << endl;
        auto node = make_shared<Node>(200);
        
        SharedResource res1(node);
        SharedResource res2(node);
        SharedResource res3(node);
        
        res1.display();
        res2.display();
        res3.display();
    }
    
    // 3. 容器中使用
    {
        cout << "\n【示例3】容器中使用" << endl;
        auto node = make_shared<Node>(300);
        
        vector<shared_ptr<Node>> vec;
        vec.push_back(node);
        vec.push_back(node);
        vec.push_back(node);
        
        cout << "引用计数：" << node.use_count() << endl;  // 4（1个原始+3个容器中）
    }
    
    cout << "\n程序结束" << endl;
    return 0;
}

### 任务4：学习 weak_ptr（预计30分钟）

**什么是 weak_ptr？**

`weak_ptr` 是弱引用智能指针：
- 不增加引用计数
- 用于打破 shared_ptr 的循环引用
- 使用前需要转换为 shared_ptr

**循环引用问题**：

```cpp
class B;  // 前向声明

class A {
public:
    shared_ptr<B> ptrB;
    ~A() { cout << "A 销毁" << endl; }
};

class B {
public:
    shared_ptr<A> ptrA;  // 循环引用！
    ~B() { cout << "B 销毁" << endl; }
};

int main() {
    auto a = make_shared<A>();
    auto b = make_shared<B>();
    
    a->ptrB = b;
    b->ptrA = a;  // 循环引用，内存泄漏！
    
    // a 和 b 永远不会被销毁
    return 0;
}
```

**使用 weak_ptr 解决**：

```cpp
class B;

class A {
public:
    shared_ptr<B> ptrB;
    ~A() { cout << "A 销毁" << endl; }
};

class B {
public:
    weak_ptr<A> ptrA;  // 使用 weak_ptr
    ~B() { cout << "B 销毁" << endl; }
};

int main() {
    auto a = make_shared<A>();
    auto b = make_shared<B>();
    
    a->ptrB = b;
    b->ptrA = a;  // 不会增加引用计数
    
    // a 和 b 正常销毁
    return 0;
}
```

**weak_ptr 的使用**：

```cpp
// 从 shared_ptr 创建 weak_ptr
shared_ptr<int> sp = make_shared<int>(100);
weak_ptr<int> wp = sp;

// weak_ptr 不增加引用计数
cout << sp.use_count() << endl;  // 1

// 使用 weak_ptr（需要先转换为 shared_ptr）
if (auto sp2 = wp.lock()) {  // lock() 返回 shared_ptr
    cout << *sp2 << endl;
} else {
    cout << "对象已被销毁" << endl;
}

// 检查对象是否还存在
if (wp.expired()) {
    cout << "对象已被销毁" << endl;
}
```

**实践练习**：

创建文件 `cpp_basics/78_weak_ptr_basics.cpp`：
```cpp
#include <memory>
#include <iostream>
using namespace std;

class Person;

class Company {
private:
    string name;
public:
    weak_ptr<Person> ceo;  // 使用 weak_ptr 避免循环引用
    
    Company(const string& n) : name(n) {
        cout << "Company " << name << " 创建" << endl;
    }
    
    ~Company() {
        cout << "Company " << name << " 销毁" << endl;
    }
    
    void showCEO() {
        if (auto p = ceo.lock()) {
            cout << name << " 的 CEO 还在" << endl;
        } else {
            cout << name << " 的 CEO 已离职" << endl;
        }
    }
};

class Person {
private:
    string name;
public:
    shared_ptr<Company> company;
    
    Person(const string& n) : name(n) {
        cout << "Person " << name << " 创建" << endl;
    }
    
    ~Person() {
        cout << "Person " << name << " 销毁" << endl;
    }
};

int main() {
    cout << "=== weak_ptr 基础示例 ===" << endl;
    
    {
        auto company = make_shared<Company>("科技公司");
        auto person = make_shared<Person>("张三");
        
        company->ceo = person;
        person->company = company;
        
        cout << "\n引用计数：" << endl;
        cout << "company: " << company.use_count() << endl;  // 2
        cout << "person: " << person.use_count() << endl;    // 1（weak_ptr不增加）
        
        company->showCEO();
    }  // 正常销毁，没有内存泄漏
    
    cout << "\n程序结束" << endl;
    return 0;
}
```
