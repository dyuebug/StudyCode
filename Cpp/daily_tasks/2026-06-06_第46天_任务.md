**核心代码预览**：
```cpp
// 事务：要么全部成功，要么全部回滚
QSqlDatabase::database().transaction();
for (auto &item : items) {
    if (!query.exec()) {
        QSqlDatabase::database().rollback();  // 失败则回滚
        return;
    }
}
QSqlDatabase::database().commit();  // 全部成功则提交

// JOIN 查询
query.exec(
    "SELECT c.name, g.group_name "
    "FROM contacts c "
    "INNER JOIN groups g ON c.group_id = g.id"
);
```
