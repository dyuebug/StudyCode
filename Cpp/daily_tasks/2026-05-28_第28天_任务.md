```cpp
    continue;
}

// 解析数据
stringstream ss(line);

getline(ss, ageStr, ',');

int age = stoi(ageStr);
if (!isValidStudentId(id) || !isValidAge(age)) {

addStudent(student);
```

### 数据验证参考

**学号验证**：
```cpp
bool StudentManager::isValidStudentId(const string& id) const {
    // 检查长度
    if (id.length() != 7) {
        return false;
    }
    
    // 检查是否全是数字
    for (char c : id) {
        if (!isdigit(c)) {
            return false;
        }
    }
    
    return true;
}
```

**年龄验证**：
```cpp
bool StudentManager::isValidAge(int age) const {
    return age >= 15 && age <= 30;
}
```

**成绩验证**：
```cpp
bool StudentManager::isValidScore(double score) const {
    return score >= 0.0 && score <= 100.0;
}
```
