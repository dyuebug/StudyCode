**任务内容**：
1. 分析性能瓶颈

   - 优化查找算法

2. 优化数据结构
   - 考虑使用 `unordered_map` 加速学号查找
   - 减少不必要的容器拷贝
   - 使用引用传递大对象

3. 优化文件操作
   - 批量写入而非逐行写入
   - 使用缓冲区
   - 减少文件打开/关闭次数

**优化示例**：
```cpp
// 优化前：线性查找 O(n)
Student* findStudentById(const string& id) {
    for (auto& student : students) {
        if (student.getId() == id) {
            return &student;
        }
    }
    return nullptr;
}

// 优化后：使用 unordered_map O(1)
// 在 StudentManager 中添加：
// unordered_map<string, Student> studentMap;
Student* findStudentById(const string& id) {
    auto it = studentMap.find(id);
    if (it != studentMap.end()) {
        return &(it->second);
    }
    return nullptr;
}
```

```cpp
// test_student_manager.cpp
#include "StudentManager.h"
#include <cassert>
#include <iostream>
using namespace std;

void testAddStudent() {
    StudentManager manager;
    Student s("2021001", "张三", 20, "计算机");
    
    assert(manager.addStudent(s) == true);
    assert(manager.getStudentCount() == 1);
    
    // 测试重复学号
    assert(manager.addStudent(s) == false);
    assert(manager.getStudentCount() == 1);
    
    cout << "✓ testAddStudent 通过" << endl;
}

void testInvalidAge() {
    StudentManager manager;
    Student s1("2021001", "测试", 10, "计算机");  // 年龄太小
    Student s2("2021002", "测试", 35, "计算机");  // 年龄太大
    
    assert(manager.addStudent(s1) == false);
    assert(manager.addStudent(s2) == false);
    assert(manager.getStudentCount() == 0);
    
    cout << "✓ testInvalidAge 通过" << endl;
}

void testSaveAndLoad() {
    StudentManager manager;
    Student s("2021001", "张三", 20, "计算机");
    manager.addStudent(s);
    
    // 保存
    assert(manager.saveToFile("test_data.txt") == true);
    
    // 清空并重新加载
    manager.clear();
    assert(manager.getStudentCount() == 0);
    
    assert(manager.loadFromFile("test_data.txt") == true);
    assert(manager.getStudentCount() == 1);
    
    cout << "✓ testSaveAndLoad 通过" << endl;
}

int main() {
    cout << "=== 运行单元测试 ===" << endl;
    
    testAddStudent();
    testInvalidAge();
    testSaveAndLoad();
    
    cout << "\n所有测试通过！✓" << endl;
    return 0;
}
```
