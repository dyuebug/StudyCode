**字数统计**：
```cpp
connect(m_editor, &QTextEdit::textChanged, this, &Notepad::updateStatusBar);
void updateStatusBar() {
    int chars = m_editor->toPlainText().length();
    int lines = m_editor->document()->lineCount();
    statusBar()->showMessage(QString("字数：%1  行数：%2").arg(chars).arg(lines));
}
```

**关闭前确认**：
```cpp
void closeEvent(QCloseEvent *event) {
    if (m_editor->document()->isModified()) {
        auto reply = QMessageBox::question(this, "未保存", "文件已修改，是否保存？",
            QMessageBox::Save | QMessageBox::Discard | QMessageBox::Cancel);
        if (reply == QMessageBox::Save) { save(); event->accept(); }
        else if (reply == QMessageBox::Cancel) { event->ignore(); }
        else { event->accept(); }
    }
}
```
