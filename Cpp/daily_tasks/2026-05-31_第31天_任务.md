### 任务1：深入理解移动语义（预计40分钟）

**回顾：什么是移动语义？**

移动语义是 C++11 引入的重要特性：
- 避免不必要的深拷贝
- 转移资源所有权而非复制
- 提高程序性能

**左值 vs 右值**：

```cpp
int a = 10;        // a 是左值（有名字，可取地址）
int b = a + 5;     // a+5 是右值（临时值，不可取地址）

string s1 = "hello";
string s2 = s1;           // 拷贝（s1 是左值）
string s3 = s1 + " world"; // 移动（s1+" world" 是右值）
```

**右值引用**：

```cpp
int&& rref = 10;           // 右值引用
string&& sref = string("hello");  // 右值引用

// 左值引用 vs 右值引用
int x = 10;
int& lref = x;             // 左值引用（绑定到左值）
int&& rref = 10;           // 右值引用（绑定到右值）
// int&& rref2 = x;        // 错误！不能绑定到左值
```

**std::move 的作用**：

```cpp
string s1 = "hello";
string s2 = std::move(s1);  // 将左值转换为右值引用

// std::move 不移动任何东西！
// 它只是类型转换：左值 → 右值引用
// 真正的移动发生在移动构造函数中
```

**移动构造函数**：

```cpp
class MyString {
private:
    char* data;
    size_t size;
public:
    // 拷贝构造函数（深拷贝）
    MyString(const MyString& other) {
        size = other.size;
        data = new char[size + 1];
        strcpy(data, other.data);
        cout << "拷贝构造" << endl;
    }
    
    // 移动构造函数（转移所有权）
    MyString(MyString&& other) noexcept {
        data = other.data;      // 转移指针
        size = other.size;
        other.data = nullptr;   // 清空源对象
        other.size = 0;
        cout << "移动构造" << endl;
    }
};
```

**移动赋值运算符**：

```cpp
// 移动赋值运算符
MyString& operator=(MyString&& other) noexcept {
    if (this != &other) {
        delete[] data;          // 释放当前资源
        
        data = other.data;      // 转移所有权
        size = other.size;
        
        other.data = nullptr;   // 清空源对象
        other.size = 0;
    }
    return *this;
}
```

**五法则（Rule of Five）**：

如果类需要自定义以下任何一个，通常需要自定义全部五个：
1. 析构函数
2. 拷贝构造函数
3. 拷贝赋值运算符
4. 移动构造函数
5. 移动赋值运算符

**noexcept 的重要性**：

```cpp
// 移动操作应该标记为 noexcept
MyString(MyString&& other) noexcept { ... }
MyString& operator=(MyString&& other) noexcept { ... }

// 原因：
// 1. STL 容器只有在移动操作不抛异常时才会使用移动
// 2. 否则会退化为拷贝（性能损失）
```

**实践练习**：

创建文件 `cpp_basics/79_move_semantics_deep.cpp`：
```cpp
#include <iostream>
#include <vector>
#include <cstring>
using namespace std;

class Buffer {
private:
    char* data;
    size_t size;
public:
    // 构造函数
    Buffer(size_t s) : size(s) {
        data = new char[size];
        cout << "构造 Buffer，大小：" << size << endl;
    }
    
    // 析构函数
    ~Buffer() {
        delete[] data;
        cout << "析构 Buffer" << endl;
    }
    
    // 拷贝构造函数
    Buffer(const Buffer& other) : size(other.size) {
        data = new char[size];
        memcpy(data, other.data, size);
        cout << "拷贝构造 Buffer，大小：" << size << endl;
    }
    
    // 拷贝赋值运算符
    Buffer& operator=(const Buffer& other) {
        if (this != &other) {
            delete[] data;
            size = other.size;
            data = new char[size];
            memcpy(data, other.data, size);
            cout << "拷贝赋值 Buffer，大小：" << size << endl;
        }
        return *this;
    }
    
    // 移动构造函数
    Buffer(Buffer&& other) noexcept : data(other.data), size(other.size) {
        other.data = nullptr;
        other.size = 0;
        cout << "移动构造 Buffer" << endl;
    }
    
    // 移动赋值运算符
    Buffer& operator=(Buffer&& other) noexcept {
        if (this != &other) {
            delete[] data;
            data = other.data;
            size = other.size;
            other.data = nullptr;
            other.size = 0;
            cout << "移动赋值 Buffer" << endl;
        }
        return *this;
    }
    
    size_t getSize() const { return size; }
};

int main() {
    cout << "=== 移动语义深入示例 ===" << endl;
    
    // 示例1：拷贝 vs 移动
    {
        cout << "\n【示例1】拷贝 vs 移动" << endl;
        
        Buffer b1(1024);
        
        Buffer b2 = b1;              // 拷贝构造
        Buffer b3 = move(b1);        // 移动构造
        
        cout << "b1 大小：" << b1.getSize() << endl;  // 0（已移动）
        cout << "b2 大小：" << b2.getSize() << endl;  // 1024
        cout << "b3 大小：" << b3.getSize() << endl;  // 1024
    }
    
    // 示例2：容器中的移动
    {
        cout << "\n【示例2】容器中的移动" << endl;
        
        vector<Buffer> vec;
        vec.reserve(3);  // 预分配空间，避免扩容
        
        cout << "添加元素..." << endl;
        vec.push_back(Buffer(512));   // 移动（临时对象）
        
        Buffer b(1024);
        vec.push_back(b);             // 拷贝（左值）
        vec.push_back(move(b));       // 移动（显式转换）
    }
    
    // 示例3：函数返回值优化
    {
        cout << "\n【示例3】函数返回值优化" << endl;
        
        auto createBuffer = []() {
            return Buffer(2048);  // RVO（返回值优化）
        };
        
        Buffer b = createBuffer();  // 通常不会调用移动构造
    }
    
    return 0;
}
```

### 任务2：学习完美转发（预计40分钟）

**什么是完美转发？**

完美转发（Perfect Forwarding）是指：
- 在函数模板中，将参数原封不动地转发给另一个函数
- 保持参数的左值/右值属性
- 保持参数的 const/非const 属性

**问题场景**：

```cpp
// 包装函数：想要转发参数给目标函数
template<typename T>
void wrapper(T arg) {
    target(arg);  // 问题：arg 总是左值！
}

string s = "hello";
wrapper(s);              // OK：s 是左值
wrapper(string("hi"));   // 问题：临时对象变成左值，无法移动
```

**解决方案：std::forward**

```cpp
template<typename T>
void wrapper(T&& arg) {  // 万能引用（Universal Reference）
    target(std::forward<T>(arg));  // 完美转发
}

// std::forward 的作用：
// - 如果 arg 绑定到左值，转发为左值引用
// - 如果 arg 绑定到右值，转发为右值引用
```

**万能引用（Universal Reference）**：

```cpp
template<typename T>
void func(T&& param);  // 万能引用（不是右值引用！）

// 规则：
// - 如果传入左值，T 推导为 T&，param 类型为 T& &&（折叠为 T&）
// - 如果传入右值，T 推导为 T，param 类型为 T&&

int x = 10;
func(x);        // T = int&，param 类型为 int&
func(10);       // T = int，param 类型为 int&&
```

**引用折叠规则**：

```cpp
// C++ 的引用折叠规则：
T& &   → T&    // 左值引用 + 左值引用 = 左值引用
T& &&  → T&    // 左值引用 + 右值引用 = 左值引用
T&& &  → T&    // 右值引用 + 左值引用 = 左值引用
T&& && → T&&   // 右值引用 + 右值引用 = 右值引用

// 简单记忆：只要有左值引用，结果就是左值引用
```

**std::forward 的实现原理**：

```cpp
// 简化版实现
template<typename T>
T&& forward(remove_reference_t<T>& arg) {
    return static_cast<T&&>(arg);
}

// 工作原理：
// 1. 如果 T = int&，返回 int& &&（折叠为 int&）
// 2. 如果 T = int，返回 int&&
```

**完美转发的应用场景**：

1. **工厂函数**：
```cpp
template<typename T, typename... Args>
unique_ptr<T> make_unique(Args&&... args) {
    return unique_ptr<T>(new T(std::forward<Args>(args)...));
}
```

2. **包装器**：
```cpp
template<typename Func, typename... Args>
auto wrapper(Func&& func, Args&&... args) {
    // 做一些额外工作
    return std::forward<Func>(func)(std::forward<Args>(args)...);
}
```

3. **容器的 emplace 方法**：
```cpp
template<typename... Args>
void emplace_back(Args&&... args) {
    // 在容器内部直接构造对象
    new (ptr) T(std::forward<Args>(args)...);
}
```

**实践练习**：

创建文件 `cpp_basics/80_perfect_forwarding.cpp`：
```cpp
#include <iostream>
#include <memory>
#include <string>
using namespace std;

// 目标函数（重载版本）
void process(int& x) {
    cout << "处理左值：" << x << endl;
}

void process(int&& x) {
    cout << "处理右值：" << x << endl;
}

// 不完美的转发
template<typename T>
void badWrapper(T arg) {
    cout << "不完美转发：";
    process(arg);  // arg 总是左值
}

// 完美转发
template<typename T>
void goodWrapper(T&& arg) {
    cout << "完美转发：";
    process(std::forward<T>(arg));
}

// 工厂函数示例
class Widget {
private:
    string name;
    int value;
public:
    Widget(const string& n, int v) : name(n), value(v) {
        cout << "Widget 构造：" << name << ", " << value << endl;
    }
    
    Widget(string&& n, int v) : name(move(n)), value(v) {
        cout << "Widget 移动构造：" << name << ", " << value << endl;
    }
};

template<typename... Args>
unique_ptr<Widget> makeWidget(Args&&... args) {
    return make_unique<Widget>(std::forward<Args>(args)...);
}

int main() {
    cout << "=== 完美转发示例 ===" << endl;
    
    // 示例1：不完美转发 vs 完美转发
    {
        cout << "\n【示例1】不完美转发 vs 完美转发" << endl;
        
        int x = 10;
        
        cout << "传入左值：" << endl;
        badWrapper(x);
        goodWrapper(x);
        
        cout << "\n传入右值：" << endl;
        badWrapper(20);
        goodWrapper(20);
    }
    
    // 示例2：工厂函数
    {
        cout << "\n【示例2】工厂函数" << endl;
        
        string name = "Widget1";
        auto w1 = makeWidget(name, 100);        // 拷贝
        auto w2 = makeWidget("Widget2", 200);   // 移动
    }
    
    return 0;
}
```

### 任务3：实现支持移动语义的类（预计40分钟）

**目标**：实现一个完整的资源管理类，支持：
- 五法则（析构、拷贝构造、拷贝赋值、移动构造、移动赋值）
- noexcept 保证
- 自我赋值检查

创建文件 `cpp_basics/81_move_class_implementation.cpp`：
```cpp
#include <iostream>
#include <cstring>
#include <algorithm>
using namespace std;

class String {
private:
    char* data;
    size_t length;
    
public:
    // 构造函数
    String(const char* str = "") {
        length = strlen(str);
        data = new char[length + 1];
        strcpy(data, str);
        cout << "构造：\"" << data << "\"" << endl;
    }
    
    // 析构函数
    ~String() {
        if (data) {
            cout << "析构：\"" << data << "\"" << endl;
            delete[] data;
        }
    }
    
    // 拷贝构造函数
    String(const String& other) {
        length = other.length;
        data = new char[length + 1];
        strcpy(data, other.data);
        cout << "拷贝构造：\"" << data << "\"" << endl;
    }
    
    // 拷贝赋值运算符
    String& operator=(const String& other) {
        if (this != &other) {  // 自我赋值检查
            delete[] data;
            length = other.length;
            data = new char[length + 1];
            strcpy(data, other.data);
            cout << "拷贝赋值：\"" << data << "\"" << endl;
        }
        return *this;
    }
    
    // 移动构造函数
    String(String&& other) noexcept 
        : data(other.data), length(other.length) {
        other.data = nullptr;
        other.length = 0;
        cout << "移动构造：\"" << data << "\"" << endl;
    }
    
    // 移动赋值运算符
    String& operator=(String&& other) noexcept {
        if (this != &other) {  // 自我赋值检查
            delete[] data;
            data = other.data;
            length = other.length;
            other.data = nullptr;
            other.length = 0;
            cout << "移动赋值：\"" << data << "\"" << endl;
        }
        return *this;
    }
    
    // 辅助方法
    const char* c_str() const { return data ? data : ""; }
    size_t size() const { return length; }
    bool empty() const { return length == 0; }
};

// 工厂函数
String createString(const char* str) {
    return String(str);  // RVO
}

int main() {
    cout << "=== 移动语义类实现示例 ===" << endl;
    
    // 示例1：基本操作
    {
        cout << "\n【示例1】基本操作" << endl;
        
        String s1("Hello");
        String s2 = s1;              // 拷贝构造
        String s3 = move(s1);        // 移动构造
        
        cout << "s1: \"" << s1.c_str() << "\"" << endl;
        cout << "s2: \"" << s2.c_str() << "\"" << endl;
        cout << "s3: \"" << s3.c_str() << "\"" << endl;
    }
    
    // 示例2：赋值操作
    {
        cout << "\n【示例2】赋值操作" << endl;
        
        String s1("World");
        String s2("C++");
        
        s2 = s1;                     // 拷贝赋值
        
        String s3("Move");
        s3 = move(s1);               // 移动赋值
    }
    
    // 示例3：函数返回
    {
        cout << "\n【示例3】函数返回" << endl;
        
        String s = createString("Factory");
    }
    
    return 0;
}
```
